# Traitors v1 (alpha) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An async Traitors game (Godot 4 web client on GitHub Pages, Supabase backend) with Google login, lobby, role dealing, scheduled banishment/murder events, a traitors-only chat, a game-master panel and avatars.

**Architecture:** All game logic lives in Postgres: RLS hides secrets, security-definer functions are the only write path, and a per-minute `pg_cron` job (`advance_games`) opens/closes/reveals scheduled events. The Godot client renders state by polling REST every 10s and calls RPCs for actions. Screens are built in GDScript code (no hand-edited `.tscn` except the tiny main scene).

**Tech Stack:** Supabase (Postgres, Auth, pg_cron, pgTAP via `supabase test db`), Godot 4.3 (GDScript, single-threaded web export), GitHub Actions + Pages, Python 3 + Pillow for the avatar import tool.

**Spec:** `docs/superpowers/specs/2026-10-12-traitors-design.md`

## Global Constraints

- Godot 4.3 web export with **thread support off** (works on plain static hosting).
- Supabase is the only backend; no other services. Client never writes tables directly, only calls RPCs.
- Google SSO only, org domain enforced server-side in `join_game` (fails closed if no domain is configured).
- Players 6–15 per game; traitors = `max(2, round(players / 4))`.
- Game master = email in `admins`; GM is never a player.
- Schedule is data (minute offsets from Monday 00:00, Europe/London). Default: Banishment Mon 09:00→Tue 15:00; Murder Tue 15:00→Thu 00:00 (reveal Thu 09:00); Banishment Wed 09:00→Thu 15:00; Murder Thu 15:00→Tue 00:00 (reveal Tue 09:00).
- Fixed times only: no early finish. Murder victim chosen at close, applied (dead + role revealed) at reveal. Banishment closes and reveals together.
- Tie → random among tied. No votes / no murder pick → nobody dies. Dead players' actions are discarded at close.
- Traitors chat only in v1; no public in-game chat. Message body 1–500 chars, 2s per-player rate limit, no BBCode rendering.
- Client polls every 10s; no realtime sockets.
- Code comments only explain *why* for unusual things.
- **Do not commit** unless the user explicitly asks. At each "Checkpoint" stop and tell the user what changed.

## Review Focus

Failure modes the spec implies; each has a pinned test in the task that owns the code.

- A player murdered/banished before an event closes still has an unrevealed vote: it must be discarded (Task 5, `04_resolution`).
- Zero votes or zero traitor picks must not crash or kill anyone (Task 5).
- Client roles calling internal functions (`advance_games`, `close_event`) must be denied (Task 6, `05_security`).
- Joining: non-org email, GM, duplicate, full game (16th), game already started must all be rejected; missing `org_domain` config must reject everyone (Task 3).
- Non-GM calling any GM function must be denied (Task 6).

```sql
create schema if not exists tests;
grant usage on schema tests to authenticated, anon;

create or replace function tests.make_user(p_name text) returns uuid language plpgsql as $$
declare v uuid := gen_random_uuid();
begin
  insert into auth.users (id, email, aud, role, raw_user_meta_data)
  values (v, p_name || case when position('@' in p_name) > 0 then '' else '@org.test' end,
          'authenticated', 'authenticated', jsonb_build_object('full_name', p_name));
  return v;
end $$;

-- security definer so a test can switch users while already running as `authenticated`
create or replace function tests.act_as(p_name text) returns void language plpgsql security definer as $$
declare u auth.users;
begin
  select * into u from auth.users
  where email = p_name || case when position('@' in p_name) > 0 then '' else '@org.test' end;
  perform set_config('request.jwt.claims', json_build_object(
    'sub', u.id, 'email', u.email, 'role', 'authenticated',
    'user_metadata', u.raw_user_meta_data)::text, true);
  perform set_config('role', 'authenticated', true);
end $$;

create or replace function tests.as_anon() returns void language plpgsql as $$
begin
  perform set_config('request.jwt.claims', '', true);
  perform set_config('role', 'anon', true);
end $$;

create or replace function tests.reset() returns void language plpgsql as $$
begin
  perform set_config('request.jwt.claims', '', true);
  execute 'reset role';
end $$;

create or replace function tests.g() returns uuid language sql stable as $$
  select current_setting('t.g')::uuid
$$;

create or replace function tests.pid(p_game uuid, p_name text) returns uuid language sql stable security definer as $$
  select id from public.players where game_id = p_game and display_name = p_name
$$;

create or replace function tests.add_players(p_game uuid, n int) returns void language plpgsql as $$
declare u uuid; i int;
begin
  for i in 1..n loop
    u := tests.make_user('p' || i);
    insert into public.players (game_id, user_id, email, display_name, avatar_key)
    values (p_game, u, 'p' || i || '@org.test', 'p' || i, 'p' || i);
  end loop;
end $$;

-- players a..n (letters), plus gm and an outsider; game already started with no events
create or replace function tests.make_game(n int) returns uuid language plpgsql as $$
declare g uuid; u uuid; i int;
begin
  insert into public.app_config values ('org_domain', 'org.test')
    on conflict (key) do update set value = excluded.value;
  insert into public.admins values ('gm@org.test') on conflict do nothing;
  perform tests.make_user('gm');
  perform tests.make_user('outsider');
  insert into public.games default values returning id into g;
  for i in 1..n loop
    u := tests.make_user(chr(96 + i));
    insert into public.players (game_id, user_id, email, display_name, avatar_key)
    values (g, u, chr(96 + i) || '@org.test', chr(96 + i), chr(96 + i));
  end loop;
  perform public.internal_start_game(g, '2026-10-19'::timestamp, '[]'::jsonb, 1, 10080);
  return g;
end $$;

create or replace function tests.set_traitors(p_game uuid, p_names text[]) returns void language sql as $$
  update public.roles r
  set role = case when p.display_name = any (p_names) then 'traitor' else 'faithful' end
  from public.players p
  where p.id = r.player_id and r.game_id = p_game
$$;

grant execute on all functions in schema tests to authenticated, anon;
```

```sql
create table app_config (key text primary key, value text not null);
create table admins (email text primary key);

create table games (
  id uuid primary key default gen_random_uuid(),
  status text not null default 'lobby' check (status in ('lobby', 'active', 'finished')),
  winner text check (winner in ('faithful', 'traitors')),
  tz text not null default 'Europe/London',
  mission jsonb,
  created_at timestamptz not null default now()
);

create table players (
  id uuid primary key default gen_random_uuid(),
  game_id uuid not null references games on delete cascade,
  user_id uuid not null references auth.users,
  email text not null,
  display_name text not null,
  avatar_key text,
  alive boolean not null default true,
  unique (game_id, user_id)
);

create table roles (
  player_id uuid primary key references players on delete cascade,
  game_id uuid not null references games on delete cascade,
  role text not null check (role in ('faithful', 'traitor'))
);

create table events (
  id uuid primary key default gen_random_uuid(),
  game_id uuid not null references games on delete cascade,
  kind text not null check (kind in ('banishment', 'murder')),
  opens_at timestamptz not null,
  closes_at timestamptz not null,
  reveals_at timestamptz not null,
  status text not null default 'scheduled' check (status in ('scheduled', 'open', 'closed', 'revealed')),
  unique (game_id, kind, opens_at)
);

-- Separate from events so a murder victim stays hidden until the event is revealed.
create table event_results (
  event_id uuid primary key references events on delete cascade,
  game_id uuid not null references games on delete cascade,
  target_id uuid references players
);

create table actions (
  id uuid primary key default gen_random_uuid(),
  event_id uuid not null references events on delete cascade,
  player_id uuid not null references players on delete cascade,
  target_id uuid references players,
  created_at timestamptz not null default now(),
  unique (event_id, player_id)
);

create table messages (
  id bigint generated always as identity primary key,
  game_id uuid not null references games on delete cascade,
  channel text not null default 'traitors' check (channel in ('traitors')),
  player_id uuid not null references players,
  body text not null check (char_length(body) between 1 and 500),
  created_at timestamptz not null default clock_timestamp()
);

alter table app_config enable row level security;
alter table admins enable row level security;
alter table games enable row level security;
alter table players enable row level security;
alter table roles enable row level security;
alter table events enable row level security;
alter table event_results enable row level security;
alter table actions enable row level security;
alter table messages enable row level security;

create function is_admin() returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from admins where email = lower(auth.jwt() ->> 'email'))
$$;

create function my_player(p_game uuid) returns uuid language sql stable security definer set search_path = public as $$
  select id from players where game_id = p_game and user_id = auth.uid()
$$;

create function in_game(p_game uuid) returns boolean language sql stable security definer set search_path = public as $$
  select my_player(p_game) is not null
$$;

create function is_traitor(p_game uuid) returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from roles r where r.player_id = my_player(p_game) and r.role = 'traitor')
$$;

-- No insert/update/delete policies anywhere: all writes go through security-definer functions.
-- app_config, admins and actions have no select policy either, so clients can never read them.
create policy games_select on games for select to authenticated using (true);
create policy players_select on players for select to authenticated using (in_game(game_id));
create policy events_select on events for select to authenticated using (in_game(game_id));

create policy roles_select on roles for select to authenticated using (
  in_game(game_id) and (
    player_id = my_player(game_id)
    or (is_traitor(game_id) and role = 'traitor')
    or exists (select 1 from players p where p.id = roles.player_id and not p.alive)
    or exists (select 1 from games g where g.id = roles.game_id and g.status = 'finished')
  )
);

create policy results_select on event_results for select to authenticated using (
  in_game(game_id) and exists (select 1 from events e where e.id = event_id and e.status = 'revealed')
);

create policy messages_select on messages for select to authenticated using (is_traitor(game_id));
```

```sql
create function am_admin() returns boolean language sql stable as $$ select is_admin() $$;

-- Offsets are minutes from Monday 00:00 local time. Murder 1 closes Wed 24:00 (= Thu 00:00);
-- murder 2 spans the weekend and closes Mon 24:00 (= Tue 00:00 next week).
create function default_schedule() returns jsonb language sql immutable as $$
  select jsonb_build_array(
    jsonb_build_object('kind', 'banishment', 'open', 0 * 1440 + 9 * 60, 'close', 1 * 1440 + 15 * 60, 'reveal', 1 * 1440 + 15 * 60),
    jsonb_build_object('kind', 'murder',     'open', 1 * 1440 + 15 * 60, 'close', 3 * 1440,           'reveal', 3 * 1440 + 9 * 60),
    jsonb_build_object('kind', 'banishment', 'open', 2 * 1440 + 9 * 60,  'close', 3 * 1440 + 15 * 60, 'reveal', 3 * 1440 + 15 * 60),
    jsonb_build_object('kind', 'murder',     'open', 3 * 1440 + 15 * 60, 'close', 8 * 1440,           'reveal', 8 * 1440 + 9 * 60))
$$;

create function create_game() returns uuid language plpgsql security definer set search_path = public as $$
declare v uuid;
begin
  if not is_admin() then raise exception 'admin only'; end if;
  insert into games default values returning id into v;
  return v;
end $$;

create function join_game(p_game uuid) returns uuid language plpgsql security definer set search_path = public as $$
declare
  v_email text := lower(auth.jwt() ->> 'email');
  v_domain text := (select value from app_config where key = 'org_domain');
  g games;
  v_id uuid;
begin
  if is_admin() then raise exception 'GM cannot play'; end if;
  select * into g from games where id = p_game;
  if g.id is null or g.status <> 'lobby' then raise exception 'game not joinable'; end if;
  if v_domain is null or v_email is null or split_part(v_email, '@', 2) <> v_domain then
    raise exception 'org accounts only';
  end if;
  if (select count(*) from players where game_id = p_game) >= 15 then raise exception 'game full'; end if;
```

```sql
create function cast_vote(p_event uuid, p_target uuid) returns void language plpgsql security definer set search_path = public as $$
declare ev events; me uuid;
begin
  select * into ev from events where id = p_event;
  if ev.id is null or ev.kind <> 'banishment' or ev.status <> 'open' or now() >= ev.closes_at then
    raise exception 'event not open';
  end if;
  me := my_player(ev.game_id);
  if me is null or not (select alive from players where id = me) then raise exception 'not allowed'; end if;
  if p_target is not null and (p_target = me or not exists (
    select 1 from players where id = p_target and game_id = ev.game_id and alive)) then
    raise exception 'invalid target';
  end if;
  insert into actions (event_id, player_id, target_id) values (p_event, me, p_target)
  on conflict (event_id, player_id) do update set target_id = excluded.target_id, created_at = now();
end $$;

create function pick_victim(p_event uuid, p_target uuid) returns void language plpgsql security definer set search_path = public as $$
declare ev events; me uuid;
begin
  select * into ev from events where id = p_event;
  if ev.id is null or ev.kind <> 'murder' or ev.status <> 'open' or now() >= ev.closes_at then
    raise exception 'event not open';
  end if;
  me := my_player(ev.game_id);
  if me is null or not is_traitor(ev.game_id) or not (select alive from players where id = me) then
    raise exception 'not allowed';
  end if;
  if p_target is not null and not exists (
    select 1 from players p join roles r on r.player_id = p.id
    where p.id = p_target and p.game_id = ev.game_id and p.alive and r.role = 'faithful') then
    raise exception 'invalid target';
  end if;
  insert into actions (event_id, player_id, target_id) values (p_event, me, p_target)
  on conflict (event_id, player_id) do update set target_id = excluded.target_id, created_at = now();

declare me uuid := my_player(p_game);
begin

     or (select status from games where id = p_game) <> 'active' then
    raise exception 'not allowed';
```

```sql
create function check_win(p_game uuid) returns void language plpgsql security definer set search_path = public as $$
declare t int; f int;
begin
  select count(*) filter (where r.role = 'traitor'), count(*) filter (where r.role = 'faithful')
  into t, f from players p join roles r on r.player_id = p.id
  where p.game_id = p_game and p.alive;
  if t = 0 then
    update games set status = 'finished', winner = 'faithful' where id = p_game;
  elsif t >= f then
    update games set status = 'finished', winner = 'traitors' where id = p_game;
  end if;
end $$;

-- Dead voters and dead targets are filtered here, not at cast time, because a murder is only
-- applied at its reveal: a vote cast before then must be discarded once the voter is dead.
create function close_event(p_event uuid) returns void language plpgsql security definer set search_path = public as $$
declare ev events; v_target uuid;
begin
  select * into ev from events where id = p_event for update;
  select a.target_id into v_target
  from actions a
  join players voter on voter.id = a.player_id and voter.alive
  join players t on t.id = a.target_id and t.alive
  where a.event_id = p_event and a.target_id is not null
  group by a.target_id
  order by count(*) desc, random()
  limit 1;
  insert into event_results (event_id, game_id, target_id) values (p_event, ev.game_id, v_target);
  update events set status = 'closed' where id = p_event;
end $$;

create function reveal_event(p_event uuid) returns void language plpgsql security definer set search_path = public as $$
declare ev events; v_target uuid;
begin
  select * into ev from events where id = p_event for update;
  select target_id into v_target from event_results where event_id = p_event;
  if v_target is not null then
    update players set alive = false where id = v_target;
  end if;
  update events set status = 'revealed' where id = p_event;
  perform check_win(ev.game_id);
end $$;

create function advance_games(p_now timestamptz default now()) returns void language plpgsql security definer set search_path = public as $$
declare e events;
begin
  update events set status = 'open'
  where status = 'scheduled' and opens_at <= p_now
    and game_id in (select id from games where status = 'active');

  for e in select ev.* from events ev join games g on g.id = ev.game_id
           where g.status = 'active' and ev.status = 'open' and ev.closes_at <= p_now
           order by ev.closes_at loop
    perform close_event(e.id);
  end loop;

  -- Re-checks game status per event: a win mid-catch-up must stop later reveals.
  for e in select ev.* from events ev join games g on g.id = ev.game_id
           where g.status = 'active' and ev.status = 'closed' and ev.reveals_at <= p_now
           order by ev.reveals_at loop
    if (select status from games where id = e.game_id) = 'active' then
      perform reveal_event(e.id);
    end if;
  end loop;
end $$;

create extension if not exists pg_cron;
select cron.schedule('advance-games', '* * * * *', $$select public.advance_games()$$);
```

```gdscript
extends Node

signal signed_in

const STORE_KEY := "traitors_session"

var access_token := ""
var refresh_token := ""
var user_email := ""

static func parse_fragment(fragment: String) -> Dictionary:
	var out := {}
	if fragment.strip_edges() == "":
		return out
	for pair in fragment.trim_prefix("#").split("&"):
		var kv := pair.split("=", true, 1)
		if kv.size() == 2:
			out[kv[0]] = kv[1].uri_decode()
	return out

static func jwt_claims(token: String) -> Dictionary:
	var parts := token.split(".")
	if parts.size() < 2:
		return {}
	# JWTs are base64url without padding; Godot's decoder wants standard base64.
	var b64 := parts[1].replace("-", "+").replace("_", "/")
	while b64.length() % 4 != 0:
		b64 += "="
	var parsed = JSON.parse_string(Marshalls.base64_to_raw(b64).get_string_from_utf8())
	return parsed if parsed is Dictionary else {}

func is_signed_in() -> bool:
	return access_token != ""

func _ready() -> void:
	var dev := OS.get_environment("TRAITORS_JWT")
	if dev != "":
		_set_session(dev, "")
		return
	if not OS.has_feature("web"):
		return
	var frag: Dictionary = parse_fragment(str(JavaScriptBridge.eval("window.location.hash")))
	if frag.has("access_token"):
		_set_session(frag["access_token"], frag.get("refresh_token", ""))
		JavaScriptBridge.eval("history.replaceState(null, '', window.location.pathname)")
		return
	var stored = JSON.parse_string(str(JavaScriptBridge.eval("localStorage.getItem('%s')" % STORE_KEY)))
	if stored is Dictionary:
		_set_session(stored.get("access_token", ""), stored.get("refresh_token", ""))
		if _expired():
			await refresh()

func _expired() -> bool:
	var exp: float = jwt_claims(access_token).get("exp", 0.0)
	return Time.get_unix_time_from_system() >= exp - 30.0

func _set_session(access: String, refresh_tok: String) -> void:
	access_token = access
	refresh_token = refresh_tok
	user_email = str(jwt_claims(access).get("email", ""))
	if OS.has_feature("web") and access != "":
		var blob := JSON.stringify({"access_token": access, "refresh_token": refresh_tok})
		JavaScriptBridge.eval("localStorage.setItem('%s', %s)" % [STORE_KEY, JSON.stringify(blob)])
	if access != "":
		signed_in.emit()

func sign_in() -> void:
	if not OS.has_feature("web"):
		push_error("Google sign-in only works in the web build; set TRAITORS_JWT for editor runs.")
		return
	var here := str(JavaScriptBridge.eval("window.location.origin + window.location.pathname"))
	var url := "%s/auth/v1/authorize?provider=google&redirect_to=%s" % [Config.SUPABASE_URL, here.uri_encode()]
	JavaScriptBridge.eval("window.location.href = '%s'" % url)

func refresh() -> bool:
	if refresh_token == "":
		return false
	var http := HTTPRequest.new()
	add_child(http)
	http.request(Config.SUPABASE_URL + "/auth/v1/token?grant_type=refresh_token",
		PackedStringArray(["apikey: " + Config.ANON_KEY, "Content-Type: application/json"]),
		HTTPClient.METHOD_POST, JSON.stringify({"refresh_token": refresh_token}))
	var res = await http.request_completed
	http.queue_free()
	var data = JSON.parse_string(res[3].get_string_from_utf8())
	if res[1] != 200 or not (data is Dictionary):
		access_token = ""
		return false
	_set_session(data["access_token"], data["refresh_token"])
	return true
```

```gdscript
func _headers() -> PackedStringArray:
	var bearer := Auth.access_token if Auth.is_signed_in() else Config.ANON_KEY

		"apikey: " + Config.ANON_KEY,

func request(method: int, path: String, body = null, retried := false) -> Dictionary:
	var http := HTTPRequest.new()
	add_child(http)
	var payload := "" if body == null else JSON.stringify(body)
	if http.request(Config.SUPABASE_URL + path, _headers(), method, payload) != OK:
		http.queue_free()
		return {"ok": false, "status": 0, "data": null}
	var res = await http.request_completed
	http.queue_free()

		return await request(method, path, body, true)
	return {
		"ok": res[1] >= 200 and res[1] < 300,
		"status": res[1],
		"data": JSON.parse_string(text) if text != "" else null,
	}

	return await request(HTTPClient.METHOD_GET, "/rest/v1/%s?%s" % [table, query])
```

```gdscript
			var b := Button.new()
			b.text = "Force close %s" % ev["kind"]
			b.pressed.connect(func():
				await Api.rpc("gm_force_close", {"p_event": ev["id"]})
				State.refresh())
			_gm.add_child(b)

func _event_panel(ev: Dictionary) -> Control:
	var box := VBoxContainer.new()
	var title := Label.new()
	title.text = "%s: closes %s" % [ev["kind"].capitalize(), ev["closes_at"].replace("T", " ").left(16)]
	box.add_child(title)
	var alive: bool = State.me.get("alive", false)
	var can_act: bool = alive and (ev["kind"] == "banishment" or State.my_role == "traitor")
	if not can_act:
		return box
	var pick := OptionButton.new()
	pick.add_item("(abstain)")
	pick.set_item_metadata(0, null)
	for p in State.players:
		var ok: bool = p["alive"] and p["id"] != State.me["id"]
		if ev["kind"] == "murder":
			ok = ok and State.role_of(p["id"]) != "traitor"
		if ok:
			pick.add_item(p["display_name"])
			pick.set_item_metadata(pick.item_count - 1, p["id"])
	box.add_child(pick)
	var btn := Button.new()
	btn.text = "Lock in"
	btn.pressed.connect(func():
		var fn := "cast_vote" if ev["kind"] == "banishment" else "pick_victim"
		await Api.rpc(fn, {"p_event": ev["id"], "p_target": pick.get_item_metadata(pick.selected)})
		State.refresh())
	box.add_child(btn)
	return box
```
Murder target filter relies on `State.role_of`, which for a traitor includes fellow traitors (visible) and unknown faithful (empty string), so faithful targets pass and traitors are excluded.
